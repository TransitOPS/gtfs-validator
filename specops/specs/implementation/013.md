## 3. Notice Definitions

### `foreign_key_violation`

Severity: **ERROR**

Emitted when `from_network_id` or `to_network_id` cannot be resolved in
either `networks.txt` or `routes.txt`.

```python
Notice(
    code="foreign_key_violation",
    severity=Severity.ERROR,
    fields={
        "child_filename": "fare_leg_join_rules.txt",
        "child_field_name": "from_network_id",   # or "to_network_id"
        "parent_filename": "routes.txt or networks.txt",
        "parent_field_name": "network_id",
        "field_value": <str>,        # the offending ID value
        "csv_row_number": <int>,     # 1-based row number from the DataFrame column
    },
)
```

**Critical wire-format note:** The `parent_filename` value is the literal
string `"routes.txt or networks.txt"`. This matches Java's string
concatenation `GtfsRoute.FILENAME + " or " + GtfsNetwork.FILENAME`. The
Python implementation must reproduce this exact string — do not swap the
order or change the separator.

**`csv_row_number` sourcing:** The analysis spec refers to `csv_row_number`
as the row number of the offending record. The `fare_leg_join_rules.txt`
schema in `schemas.py` does not declare a `csv_row_number` column — this
value must be derived at load time using `with_row_index`. The loading
convention used by other validators (e.g., `fare_attribute_agency_id.py`)
uses `with_row_index("_row_idx")` to attach a 0-based integer index. The
spec test contracts from the Java source reference row numbers starting at
1 for the first data row; confirm against the feed loading convention before
emitting. Use the same `_row_idx` pattern as existing validators and match
whatever the project-wide convention is for `csv_row_number` (0-based data
index).

### `missing_required_field`

Severity: **ERROR**

Emitted when exactly one of `from_stop_id` / `to_stop_id` is present and
the other is absent.

```python
Notice(
    code="missing_required_field",
    severity=Severity.ERROR,
    fields={
        "filename": "fare_leg_join_rules.txt",
        "csv_row_number": <int>,     # same row index convention as above
        "field_name": "to_stop_id",  # or "from_stop_id" — whichever is missing
    },
)
```

## 4. Implementation Approach

```
function validate_fare_leg_join_rule(feed, ctx):

    # --- Skip guard ---
    if "fare_leg_join_rules" not in feed or feed["fare_leg_join_rules"].is_empty():
        return []

    df = feed["fare_leg_join_rules"].with_row_index("_row_idx")

    # --- Build valid network ID lookup set (OR across networks.txt and routes.txt) ---
    valid_network_ids: set[str] = set()
    if "networks" in feed:
        col = feed["networks"]["network_id"].drop_nulls()
        valid_network_ids |= set(col.to_list())
    if "routes" in feed and "network_id" in feed["routes"].columns:
        col = feed["routes"]["network_id"].drop_nulls().filter(
            feed["routes"]["network_id"].drop_nulls() != ""
        )
        # Simpler: use Series.filter with is_not_null + != ""
        routes_col = feed["routes"]["network_id"]
        valid_network_ids |= set(
            routes_col.filter(
                routes_col.is_not_null() & (routes_col != "")
            ).to_list()
        )

    notices: list[Notice] = []

    # --- Check A & B: from_network_id and to_network_id FK ---
    for field_name in ("from_network_id", "to_network_id"):
        if field_name not in df.columns:
            continue

        # Rows where the field is present (non-null, non-empty) AND not in valid set
        violations = df.filter(
            pl.col(field_name).is_not_null()
            & (pl.col(field_name) != "")
            & ~pl.col(field_name).is_in(list(valid_network_ids))
        )

        for row in violations.iter_rows(named=True):
            notices.append(
                Notice(
                    code="foreign_key_violation",
                    severity=Severity.ERROR,
                    fields={
                        "child_filename": "fare_leg_join_rules.txt",
                        "child_field_name": field_name,
                        "parent_filename": "routes.txt or networks.txt",
                        "parent_field_name": "network_id",
                        "field_value": row[field_name],
                        "csv_row_number": row["_row_idx"],
                    },
                )
            )

    # --- Check C: from_stop_id / to_stop_id co-presence ---
    has_from_col = "from_stop_id" in df.columns
    has_to_col = "to_stop_id" in df.columns

    def is_present(col_name: str) -> pl.Expr:
        return pl.col(col_name).is_not_null() & (pl.col(col_name) != "")

    def is_absent(col_name: str) -> pl.Expr:
        return pl.col(col_name).is_null() | (pl.col(col_name) == "")

    # from_stop_id present, to_stop_id absent
    if has_from_col and has_to_col:
        missing_to = df.filter(is_present("from_stop_id") & is_absent("to_stop_id"))
        for row in missing_to.iter_rows(named=True):
            notices.append(
                Notice(
                    code="missing_required_field",
                    severity=Severity.ERROR,
                    fields={
                        "filename": "fare_leg_join_rules.txt",
                        "csv_row_number": row["_row_idx"],
                        "field_name": "to_stop_id",
                    },
                )
            )

        # to_stop_id present, from_stop_id absent
        missing_from = df.filter(is_present("to_stop_id") & is_absent("from_stop_id"))
        for row in missing_from.iter_rows(named=True):
            notices.append(
                Notice(
                    code="missing_required_field",
                    severity=Severity.ERROR,
                    fields={
                        "filename": "fare_leg_join_rules.txt",
                        "csv_row_number": row["_row_idx"],
                        "field_name": "from_stop_id",
                    },
                )
            )
    elif has_from_col and not has_to_col:
        # to_stop_id column absent entirely: any row with from_stop_id present is a violation
        missing_to = df.filter(is_present("from_stop_id"))
        for row in missing_to.iter_rows(named=True):
            notices.append(
                Notice(
                    code="missing_required_field",
                    severity=Severity.ERROR,
                    fields={
                        "filename": "fare_leg_join_rules.txt",
                        "csv_row_number": row["_row_idx"],
                        "field_name": "to_stop_id",
                    },
                )
            )
    elif has_to_col and not has_from_col:
        # from_stop_id column absent entirely: any row with to_stop_id present is a violation
        missing_from = df.filter(is_present("to_stop_id"))
        for row in missing_from.iter_rows(named=True):
            notices.append(
                Notice(
                    code="missing_required_field",
                    severity=Severity.ERROR,
                    fields={
                        "filename": "fare_leg_join_rules.txt",
                        "csv_row_number": row["_row_idx"],
                        "field_name": "from_stop_id",
                    },
                )
            )
    # If neither column is present: no co-presence check possible; no notices.

    return notices
```

**Key design decisions:**

- **Set-based OR lookup for network IDs.** The valid network ID set is built
  once before iterating over checks. This avoids repeated cross-joins or
  redundant lookups. Both `networks.txt` and `routes.txt` contribute to the
  same set; the OR semantics are achieved implicitly via set union.
- **Empty string treated as absent for network IDs.** The analysis spec
  section 5 edge case and section 7 note confirm that Java's `hasFromNetworkId()`
  returns false for both null and empty string. The Polars filter `is_not_null()
  & (col != "")` matches this behavior. Empty-string values do not emit an
  FK violation.
- **Empty string treated as absent for stop IDs.** Same rationale: null and
  empty string are both "absent" for the co-presence check. Use `is_null() |
  (col == "")` as the absent predicate.
- **`routes.txt.network_id` column guard.** `routes.txt` has `network_id` as
  a conditionally required column (see `schemas.py` line 242). The column may
  not be present in all feeds. Guard with `"network_id" in feed["routes"].columns`
  before accessing it to avoid a Polars `ColumnNotFoundError`.
- **Independent checks A and B.** A single row may emit both a
  `from_network_id` FK violation and a `to_network_id` FK violation. The loop
  over `("from_network_id", "to_network_id")` handles each independently.
- **Polars filter before Python iteration.** All row selection is done via
  Polars expressions. Only the (typically small) set of offending rows is
  iterated in Python, matching the pattern in `fare_attribute_agency_id.py`.
- **Column existence guard for stop ID checks.** The `fare_leg_join_rules.txt`
  schema defines `from_stop_id` and `to_stop_id` as conditionally required;
  a feed may supply the file without either column. Guard all column access
  with `in df.columns` checks.
- **`is_in` with a Python list.** When `valid_network_ids` is a Python set,
  pass `list(valid_network_ids)` to `pl.Expr.is_in`. Polars accepts a Python
  list or a Polars Series; the list form is acceptable for sets of IDs that
  are typically small (one network per route).
- **Notice ordering.** FK violation notices for `from_network_id` are emitted
  before `to_network_id` notices (loop order). Within each field, notices are
  emitted in ascending `_row_idx` order (Polars filter preserves row order).
  Stop ID notices are emitted after all FK notices. No explicit sorting is
  required.

In `src/gtfs_validator/validators/__init__.py`, add the import and entry:

```python
from gtfs_validator.validators.fare_leg_join_rule import validate_fare_leg_join_rule
```

Add to `VALIDATOR_REGISTRY` after the `"fare_attribute_agency_id"` entry
(alphabetical by `name`):

```python
ValidatorEntry(
    name="fare_leg_join_rule",
    fn=validate_fare_leg_join_rule,
    requires=["fare_leg_join_rules"],
),
```

The `requires` list contains only `"fare_leg_join_rules"` because the
validator's own skip guard handles the primary table absence. The
`networks` and `routes` tables are optional parents; the validator handles
their absence gracefully by producing no valid IDs in the lookup set (which
then causes FK violations for every non-null network ID value — the correct
behavior per the analysis spec edge cases).

def make_join_rules(**kwargs) -> pl.DataFrame:
    """Build a fare_leg_join_rules DataFrame with sensible defaults."""
    defaults = {
        "from_network_id": ["network1"],
        "to_network_id": ["network2"],
        "from_stop_id": [None],
        "to_stop_id": [None],
    }
    defaults.update(kwargs)
    return pl.DataFrame(defaults)


def make_networks(network_ids: list[str | None]) -> pl.DataFrame:
    return pl.DataFrame({"network_id": network_ids})


def make_routes(network_ids: list[str | None]) -> pl.DataFrame:
    return pl.DataFrame({"route_id": [f"r{i}" for i in range(len(network_ids))], "network_id": network_ids})
