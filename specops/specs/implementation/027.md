# Implementation: LocationHasStopTimesValidator

## 1. Module Location

**Validator:** `src/gtfs_validator/validators/location_has_stop_times.py`

This is a cross-table validator that checks the relationship between `stops.txt`, `stop_times.txt`, and optionally `location_group_stops.txt`. It ensures stops (location_type=0) have stop time references and that non-stop locations do not. The logic uses set-building and a Polars inner join.

## 2. Function Signature

```python
from __future__ import annotations

import polars as pl

from gtfs_validator.context import ValidationContext
from gtfs_validator.notices import Notice, Severity


def validate_location_has_stop_times(
    feed: dict[str, pl.DataFrame],
    ctx: ValidationContext,
) -> list[Notice]:
```

No internal helper functions are needed beyond the main validator function.

## 3. Notice Definitions

### `stop_without_stop_time`

Severity: **WARNING**

Emitted when a stop (location_type=0 or null/missing) has no stop time references, either directly via `stop_times.stop_id` or indirectly via a location group referenced in `stop_times.location_group_id`.

```python
Notice(
    code="stop_without_stop_time",
    severity=Severity.WARNING,
    fields={
        "csv_row_number": <int>,   # row number in stops.txt
        "stop_id": <str>,          # the unreferenced stop_id
        "stop_name": <str>,        # the stop_name (empty string if missing)
    },
)
```

### `location_with_unexpected_stop_time`

Severity: **ERROR**

Emitted when a non-stop location (location_type in {1, 2, 3, 4}) is directly referenced by `stop_times.stop_id`.

```python
Notice(
    code="location_with_unexpected_stop_time",
    severity=Severity.ERROR,
    fields={
        "csv_row_number": <int>,              # row number in stops.txt
        "stop_id": <str>,                     # the offending stop_id
        "stop_name": <str>,                   # the stop_name
        "stop_time_csv_row_number": <int>,    # row number of first offending stop_time
    },
)
```

## 4. Implementation Approach

### Strategy: Set-building + Polars inner join

The validator has two distinct checks with different semantics:

- **WARNING check** (stop_without_stop_time): Uses set membership to determine if a stop is referenced at all, either directly or via location groups. This is a set-based anti-membership check best done with Python sets.
- **ERROR check** (location_with_unexpected_stop_time): Uses a Polars inner join between non-stop locations and stop_times to find direct references, then groups by stop_id to get the first offending stop_time row.

### Algorithm

```
1.  stops = feed.get("stops")
2.  stop_times = feed.get("stop_times")
3.  if stops is None or stop_times is None:
4.      return []
5.  if stops.is_empty():
6.      return []
7.
8.  location_group_stops = feed.get("location_group_stops")
9.
10. # Step 1: Build the set of stop_ids referenced in stop_times
11. #   (directly via stop_id, or indirectly via location_group_id)
12. referenced_stop_ids: set[str] = set()
13. if "stop_id" in stop_times.columns:
14.     referenced_stop_ids = set(
15.         stop_times.select("stop_id").drop_nulls().to_series().to_list()
16.     )
17.
18. if (
19.     "location_group_id" in stop_times.columns
20.     and location_group_stops is not None
21.     and not location_group_stops.is_empty()
22. ):
23.     lg_ids = set(
24.         stop_times.select("location_group_id").drop_nulls().to_series().to_list()
25.     )
26.     if lg_ids:
27.         lg_stop_ids = (
28.             location_group_stops
29.             .filter(pl.col("location_group_id").is_in(list(lg_ids)))
30.             .select("stop_id")
31.             .drop_nulls()
32.             .to_series()
33.             .to_list()
34.         )
35.         referenced_stop_ids.update(lg_stop_ids)
36.
37. notices: list[Notice] = []
38.
39. # Step 2a: Stops (location_type=0 or null) without any stop time reference
40. actual_stops = stops.filter(
41.     (pl.col("location_type") == 0) | pl.col("location_type").is_null()
42. )
43. for row in actual_stops.iter_rows(named=True):
44.     if row["stop_id"] not in referenced_stop_ids:
45.         notices.append(Notice(
46.             code="stop_without_stop_time",
47.             severity=Severity.WARNING,
48.             fields={
49.                 "csv_row_number": row["csvRowNumber"],
50.                 "stop_id": row["stop_id"],
51.                 "stop_name": row.get("stop_name", "") or "",
52.             },
53.         ))
54.
55. # Step 2b: Non-stop locations with direct stop time references
56. non_stops = stops.filter(
57.     pl.col("location_type").is_not_null() & (pl.col("location_type") != 0)
58. )
59. if not non_stops.is_empty() and not stop_times.is_empty() and "stop_id" in stop_times.columns:
60.     bad = non_stops.join(
61.         stop_times.select("stop_id", "csvRowNumber").rename(
62.             {"csvRowNumber": "stop_time_csv_row_number"}
63.         ),
64.         on="stop_id",
65.         how="inner",
66.     )
67.     if not bad.is_empty():
68.         first_bad = bad.group_by("stop_id").agg([
69.             pl.col("csvRowNumber").first(),
70.             pl.col("stop_name").first(),
71.             pl.col("stop_time_csv_row_number").min(),
72.         ])
73.         for row in first_bad.iter_rows(named=True):
74.             notices.append(Notice(
75.                 code="location_with_unexpected_stop_time",
76.                 severity=Severity.ERROR,
77.                 fields={
78.                     "csv_row_number": row["csvRowNumber"],
79.                     "stop_id": row["stop_id"],
80.                     "stop_name": row["stop_name"],
81.                     "stop_time_csv_row_number": row["stop_time_csv_row_number"],
82.                 },
83.             ))
84.
85. return notices
```

### Handling missing tables and columns

- If `stops` or `stop_times` is absent from the feed dict, return early with no notices. The `requires` list in the registry entry also guards against this.
- If `location_group_stops` is absent, treat as empty -- location group membership is not considered. Only direct `stop_id` references in `stop_times` matter.
- If `stop_times` lacks the `location_group_id` column, skip location group resolution entirely.
- If `stop_times` lacks the `stop_id` column, the referenced set stays empty and the join for the ERROR check is skipped.
- If `stops` is empty, return early with no notices.

### Key design decisions

- The `location_type` column is compared as integer since the schema defines it as `_ENUM` which gets cast to integer during loading. Null values are treated as 0 (STOP), per GTFS spec default.
- For the WARNING check (`stop_without_stop_time`), a stop is considered referenced if it appears in `stop_times.stop_id` directly OR belongs to a location group referenced in `stop_times.location_group_id`. Both paths must be checked.
- For the ERROR check (`location_with_unexpected_stop_time`), only direct `stop_id` references in `stop_times.txt` trigger the error. Non-stop locations referenced indirectly via `location_group_id` do not trigger errors (matching the Java behavior where `stopTimeTable.byStopId()` indexes by `stop_id` only).
- When multiple stop_times reference the same non-stop location, only one error notice is emitted per stop, using the minimum `csvRowNumber` from stop_times as `stop_time_csv_row_number`.
- The `iter_rows` loops are acceptable because the number of problematic stops is expected to be small relative to the total dataset.
- `stop_name` defaults to empty string if null or missing.

## 5. Registry Entry

In `src/gtfs_validator/validators/__init__.py`, add the import:

```python
from gtfs_validator.validators.location_has_stop_times import (
    validate_location_has_stop_times,
)
```

Append the entry to `VALIDATOR_REGISTRY`:

```python
ValidatorEntry(
    name="location_has_stop_times",
    fn=validate_location_has_stop_times,
    requires=["stops", "stop_times"],
),
```

Note: `location_group_stops` is NOT listed in `requires` because it is optional -- the validator handles its absence gracefully. Listing only `stops` and `stop_times` ensures the validator runs whenever both required tables are present.

```python
from datetime import date
import polars as pl
import pytest
from gtfs_validator.context import ValidationContext
from gtfs_validator.notices import Severity
from gtfs_validator.validators.location_has_stop_times import (
    validate_location_has_stop_times,
)

CTX = ValidationContext(country_code="US", date_for_validation=date(2026, 3, 8))


def make_stops(rows: list[tuple[str, str | None, int | None]], row_start: int = 2) -> pl.DataFrame:
    """Each row is (stop_id, stop_name, location_type)."""
    return pl.DataFrame(
        {
            "csvRowNumber": list(range(row_start, row_start + len(rows))),
            "stop_id": [r[0] for r in rows],
            "stop_name": [r[1] for r in rows],
            "location_type": [r[2] for r in rows],
        },
        schema={
            "csvRowNumber": pl.Int64,
            "stop_id": pl.Utf8,
            "stop_name": pl.Utf8,
            "location_type": pl.Int64,
        },
    )


def make_stop_times(
    rows: list[tuple[str, str | None]],
    row_start: int = 2,
    include_location_group_id: bool = False,
) -> pl.DataFrame:
    """Each row is (stop_id, location_group_id).

    If include_location_group_id is False, the location_group_id column is omitted.
    """
    data: dict = {
        "csvRowNumber": list(range(row_start, row_start + len(rows))),
        "stop_id": [r[0] for r in rows],
    }
    if include_location_group_id:
        data["location_group_id"] = [r[1] for r in rows]
    return pl.DataFrame(data)


def make_location_group_stops(rows: list[tuple[str, str]]) -> pl.DataFrame:
    """Each row is (location_group_id, stop_id)."""
    return pl.DataFrame(
        {
            "location_group_id": [r[0] for r in rows],
            "stop_id": [r[1] for r in rows],
        }
    )
```
