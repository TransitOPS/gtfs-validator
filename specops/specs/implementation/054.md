## 4. Implementation approach

### 4.1 Language split

- **Polars**: data loading, grouping, lookup-dict construction only.
- **Pure Python**: all geometry and matching algorithm code (in `shape_to_stop_matching_util.py`).

This follows the AGENTS.md rule: "Use plain Python algorithms for graph/geometry/interval
logic when clearer and safer than forcing Polars."

### 4.2 Data classes (defined in `shape_to_stop_matching_util.py`)

```python
from dataclasses import dataclass

@dataclass
class ShapePoint:
    geo_distance: float        # cumulative great-circle distance from shape start (metres)
    user_distance: float       # running max of shape_dist_traveled (0.0 if null)
    lat: float
    lon: float

@dataclass
class StopPoint:
    lat: float
    lon: float
    user_distance: float       # stop_time.shape_dist_traveled (0.0 if null)
    stop_time_row: dict        # original stop_time row dict
    is_large_station: bool     # True only for first/last stop on RAIL routes

@dataclass
class CandidateMatch:
    geo_distance_to_shape: float   # metres from stop to matched point on shape
    shape_geo_distance: float      # ShapePoint.geo_distance at the match location
    lat: float                     # matched point lat
    lon: float                     # matched point lon

@dataclass
class Problem:
    kind: str   # "too_far" | "too_many_matches" | "out_of_order"
    # Fields vary by kind — use a dict for flexibility
    data: dict
```

### 4.4 Helper function inventory (`shape_to_stop_matching_util.py`)

#### `geo_distance_meters(lat1, lon1, lat2, lon2) -> float`
Haversine great-circle distance using `EARTH_RADIUS_METERS = 6_371_010.0`.

```
phi1, phi2 = radians(lat1), radians(lat2)
dphi = phi2 - phi1
dlambda = radians(lon2 - lon1)
a = sin(dphi/2)^2 + cos(phi1)*cos(phi2)*sin(dlambda/2)^2
return 2 * atan2(sqrt(a), sqrt(1-a)) * EARTH_RADIUS_METERS
```

#### `latlng_to_unit_vector(lat_deg, lon_deg) -> tuple[float, float, float]`
Convert geographic coordinates to a unit 3-vector on the unit sphere.

```
lat, lon = radians(lat_deg), radians(lon_deg)
return (cos(lat)*cos(lon), cos(lat)*sin(lon), sin(lat))
```

#### `unit_vector_to_latlng(v) -> tuple[float, float]`
Inverse of the above.

```
lat = degrees(asin(clamp(v[2], -1, 1)))
lon = degrees(atan2(v[1], v[0]))
return (lat, lon)
```

#### `closest_point_on_edge(p, a, b) -> tuple[float, float]`
Returns (lat, lon) of the closest point on the great-circle arc from `a` to `b`
to query point `p` (all as unit 3-vectors). Implements the S2-equivalent
spherical closest-point formula:

```
pseudocode:
  # If arc is degenerate (a == b), return a
  n = cross(a, b)                    # normal to plane of arc
  if |n| < epsilon: return unit_vector_to_latlng(a)
  # Project p onto the plane perpendicular to n
  p_proj = p - dot(p, n)*n
  p_proj = normalize(p_proj)
  # Check if p_proj lies within the arc
  if dot(cross(a, p_proj), n) >= 0 and dot(cross(p_proj, b), n) >= 0:
      return unit_vector_to_latlng(p_proj)
  # Otherwise return whichever endpoint is closer
  if dot(a, p) >= dot(b, p):
      return unit_vector_to_latlng(a)
  return unit_vector_to_latlng(b)
```

#### `build_shape_points(shape_rows: list[dict]) -> list[ShapePoint]`

```
sort shape_rows by shape_pt_sequence
geo_dist = 0.0
user_dist = 0.0
result = []
prev_lat, prev_lon = None, None
for row in sorted_rows:
    lat = row["shape_pt_lat"]
    lon = row["shape_pt_lon"]
    if prev_lat is not None:
        geo_dist += geo_distance_meters(prev_lat, prev_lon, lat, lon)
    raw_user = row.get("shape_dist_traveled") or 0.0
    user_dist = max(user_dist, raw_user)
    result.append(ShapePoint(geo_distance=geo_dist, user_distance=user_dist,
                              lat=lat, lon=lon))
    prev_lat, prev_lon = lat, lon
return result
```

#### `resolve_stop_location(stop_id: str, stops_by_id: dict) -> tuple[float, float]`

```
stop = stops_by_id.get(stop_id)
for _ in range(4):   # up to 3 parent hops (4 iterations: original + 3 parents)
    if stop is None:
        return (0.0, 0.0)
    lat = stop.get("stop_lat")
    lon = stop.get("stop_lon")
    if lat is not None and lon is not None:
        return (float(lat), float(lon))
    parent_id = stop.get("parent_station")
    if not parent_id:
        return (0.0, 0.0)
    stop = stops_by_id.get(parent_id)
return (0.0, 0.0)
```

#### `resolve_stop_name(stop_id: str, stops_by_id: dict) -> str`

```
stop = stops_by_id.get(stop_id)
if stop is None:
    return ""
return stop.get("stop_name") or ""
```

#### `build_stop_points(stop_times: list[dict], stops_by_id: dict, is_large_route: bool) -> list[StopPoint]`

```
sorted by stop_sequence
n = len(stop_times)
result = []
for i, st in enumerate(sorted_stop_times):
    lat, lon = resolve_stop_location(st["stop_id"], stops_by_id)
    user_dist = st.get("shape_dist_traveled") or 0.0
    is_large = is_large_route and (i == 0 or i == n - 1)
    result.append(StopPoint(lat=lat, lon=lon, user_distance=user_dist,
                             stop_time_row=st, is_large_station=is_large))
return result
```

#### `compute_trip_hash(stop_times: list[dict]) -> bytes`
Stable hash over `(stop_id, shape_dist_traveled)` sequence using `hashlib.sha256`.
Null `shape_dist_traveled` is treated as `0.0`.

```
import hashlib, struct
h = hashlib.sha256()
h.update(struct.pack(">I", len(stop_times)))
for st in stop_times:
    sid = (st.get("stop_id") or "").encode("utf-8")
    h.update(struct.pack(">I", len(sid)))
    h.update(sid)
    dist = st.get("shape_dist_traveled") or 0.0
    h.update(struct.pack(">d", float(dist)))
return h.digest()
```

#### `find_potential_matches(stop: StopPoint, shape_points: list[ShapePoint], max_dist: float) -> list[CandidateMatch]`
Local-minimum scan across all consecutive shape edges.

```
p = latlng_to_unit_vector(stop.lat, stop.lon)
in_close_run = False
run_best_dist = inf
run_best_match = None
matches = []

for i in range(len(shape_points) - 1):
    a = shape_points[i]
    b = shape_points[i+1]
    av = latlng_to_unit_vector(a.lat, a.lon)
    bv = latlng_to_unit_vector(b.lat, b.lon)
    closest_lat, closest_lon = closest_point_on_edge(p, av, bv)
    dist = geo_distance_meters(stop.lat, stop.lon, closest_lat, closest_lon)
    # Interpolate geo_distance at matched point along segment
    seg_frac = ... (fraction along segment)
    matched_geo_dist = a.geo_distance + seg_frac * (b.geo_distance - a.geo_distance)
    candidate = CandidateMatch(geo_distance_to_shape=dist,
                               shape_geo_distance=matched_geo_dist,
                               lat=closest_lat, lon=closest_lon)
    if dist <= max_dist:
        if not in_close_run:
            in_close_run = True
            run_best_dist = dist
            run_best_match = candidate
        elif dist < run_best_dist:
            run_best_dist = dist
            run_best_match = candidate
    else:
        if in_close_run:
            matches.append(run_best_match)
            in_close_run = False

if in_close_run:
    matches.append(run_best_match)

return matches
```

For the single-point shape edge case (shape has only one point), return
`[CandidateMatch(...)]` using that single point directly.

#### `find_closest_on_shape(stop: StopPoint, shape_points: list[ShapePoint]) -> CandidateMatch`
Scans all edges and returns the globally closest point (used when all edges are
beyond `max_distance` to report the best available match in `stop_too_far_from_shape`).

#### `find_best_assignment(candidates_per_stop: list[list[CandidateMatch]], stop_points: list[StopPoint]) -> tuple[list[CandidateMatch | None], list[tuple[int, CandidateMatch, CandidateMatch]]]`
DP assignment: for each stop, pick one candidate such that `shape_geo_distance`
values are non-decreasing and total `geo_distance_to_shape` is minimized.

```
n = len(candidates_per_stop)
# dp[i][j] = (min_total_dist, prev_idx) for stop i using candidate j
# Initialize dp[0]
for each stop i from 0 to n-1:
    for each candidate j of stop i:
        for each valid prev candidate k of stop i-1 (shape_geo_distance non-decreasing):
            update dp[i][j] if dp[i-1][k].cost + candidate[j].geo_distance_to_shape is lower

backtrack to get best assignment
for any stop with no valid prior candidate: emit out_of_order problem for that stop
```

Returns `(assignment list, list of (stop_idx, match_of_this_stop, match_of_prev_stop))`
for out-of-order problems.

#### `interpolate_user_distance(user_dist: float, shape_points: list[ShapePoint], search_from: int) -> tuple[CandidateMatch, int]`
Find the segment where `shape_point[i].user_distance <= user_dist <= shape_point[i+1].user_distance`
starting the search from `search_from` (enforcing monotonicity). Interpolate lat/lon
linearly in 3D unit-vector space, then project back to lat/lon. Return the match
and the index to continue from for the next stop.

#### `match_using_geo_distance(stop_points, shape_points, settings) -> list[Problem]`

```
for each stop in stop_points:
    max_dist = settings.max_distance_meters
    if stop.is_large_station:
        max_dist *= settings.large_station_multiplier
    candidates = find_potential_matches(stop, shape_points, max_dist)
    if len(candidates) > settings.potential_matches_threshold:
        emit STOP_HAS_TOO_MANY_MATCHES problem
    if not candidates:
        closest = find_closest_on_shape(stop, shape_points)
        emit STOP_TOO_FAR_FROM_SHAPE problem using closest
        return problems_so_far   # abort trip matching

assignment, out_of_order = find_best_assignment(candidates_per_stop, stop_points)
for (stop_idx, match1, match2) in out_of_order:
    emit STOPS_MATCH_OUT_OF_ORDER problem
return problems
```

#### `match_using_user_distance(stop_points, shape_points, settings) -> list[Problem]`

```
search_from = 0
candidates_per_stop = []
for stop in stop_points:
    if stop.user_distance > 0.0:
        match, search_from = interpolate_user_distance(stop.user_distance,
                                                        shape_points, search_from)
        candidates_per_stop.append([match])
    else:
        max_dist = settings.max_distance_meters
        if stop.is_large_station:
            max_dist *= settings.large_station_multiplier
        candidates_per_stop.append(find_potential_matches(stop, shape_points, max_dist))

assignment, out_of_order = find_best_assignment(candidates_per_stop, stop_points)

for i, match in enumerate(assignment):
    stop = stop_points[i]
    max_dist = settings.max_distance_meters
    if stop.is_large_station:
        max_dist *= settings.large_station_multiplier
    if match and match.geo_distance_to_shape > max_dist:
        emit STOP_TOO_FAR_FROM_SHAPE (user_distance variant) problem

for (stop_idx, match1, match2) in out_of_order:
    emit STOPS_MATCH_OUT_OF_ORDER problem

return problems
```

### 4.5 Top-level validator pseudocode (`shape_to_stop_matching.py`)

```
def validate_shape_to_stop_matching(feed, ctx):
    REQUIRED = ["stops", "trips", "routes", "stop_times", "shapes"]
    for table in REQUIRED:
        if table not in feed or feed[table].is_empty():
            return []

    stops_df      = feed["stops"]
    trips_df      = feed["trips"]
    routes_df     = feed["routes"]
    stop_times_df = feed["stop_times"]
    shapes_df     = feed["shapes"]

    # Build lookup dicts once (Polars → Python dicts)
    stops_by_id   = {row["stop_id"]: row
                     for row in stops_df.select(["stop_id","stop_name","stop_lat",
                                                 "stop_lon","parent_station",
                                                 "location_type"]).to_dicts()}
    routes_by_id  = {row["route_id"]: row
                     for row in routes_df.select(["route_id","route_type"]).to_dicts()}

    # Group stop_times by trip_id
    st_by_trip_id: dict[str, list[dict]] = defaultdict(list)
    for row in stop_times_df.select(["trip_id","stop_id","stop_sequence",
                                      "shape_dist_traveled","csv_row_number"]).to_dicts():
        st_by_trip_id[row["trip_id"]].append(row)
    # Sort each group by stop_sequence
    for tid in st_by_trip_id:
        st_by_trip_id[tid].sort(key=lambda r: r["stop_sequence"])

    # Group trips by shape_id
    trips_by_shape_id: dict[str, list[dict]] = defaultdict(list)
    for row in trips_df.select(["trip_id","route_id","shape_id",
                                  "csv_row_number"]).to_dicts():
        if row.get("shape_id"):
            trips_by_shape_id[row["shape_id"]].append(row)

    # Group shapes by shape_id
    shapes_groups: dict[str, list[dict]] = defaultdict(list)
    for row in shapes_df.select(["shape_id","shape_pt_lat","shape_pt_lon",
                                   "shape_pt_sequence","shape_dist_traveled"]).to_dicts():
        shapes_groups[row["shape_id"]].append(row)

    settings = MatchSettings()
    notices = []

    for shape_id, shape_rows in shapes_groups.items():
        trips_for_shape = trips_by_shape_id.get(shape_id, [])
        if not trips_for_shape:
            continue

        shape_points = build_shape_points(shape_rows)
        shape_has_user_dist = shape_points[-1].user_distance > 0.0
        reported_stop_ids: set[str] = set()

        seen_trip_hashes: set[bytes] = set()

        for trip in trips_for_shape:
            trip_id = trip["trip_id"]
            stop_times = st_by_trip_id.get(trip_id, [])
            if not stop_times:
                continue

            trip_hash = compute_trip_hash(stop_times)
            if trip_hash in seen_trip_hashes:
                continue
            seen_trip_hashes.add(trip_hash)

            route = routes_by_id.get(trip["route_id"])
            if route is None:
                continue

            is_large_route = (route["route_type"] == RAIL_ROUTE_TYPE)
            stop_points = build_stop_points(stop_times, stops_by_id, is_large_route)

            # Geo-distance matching (always)
            geo_problems = match_using_geo_distance(stop_points, shape_points, settings)
            notices.extend(
                _problems_to_notices(geo_problems, trip, shape_id, stops_by_id,
                                     reported_stop_ids,
                                     notice_code="stop_too_far_from_shape")
            )

            # User-distance matching (conditional)
            stops_have_user_dist = stop_points[-1].user_distance > 0.0
            if stops_have_user_dist and shape_has_user_dist:
                user_problems = match_using_user_distance(stop_points, shape_points,
                                                           settings)
                notices.extend(
                    _problems_to_notices(user_problems, trip, shape_id, stops_by_id,
                                         reported_stop_ids,
                                         notice_code="stop_too_far_from_shape_using_user_distance")
                )

    return notices
```

#### `_problems_to_notices` helper

Converts `Problem` objects to `Notice` objects, applying the two filtering rules:
1. Skip if `stop_time.stop_id` is empty/null.
2. For `stop_too_far_from_shape` / `stop_too_far_from_shape_using_user_distance`:
   skip if `stop_id` is already in `reported_stop_ids`; otherwise add to it.
