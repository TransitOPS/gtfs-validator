# Implementation: StopZoneIdValidator

## 1. Module Location

**Validator file:** `src/gtfs_validator/validators/stop_zone_id.py`

No shared utilities needed. This validator uses a Polars join chain across
four tables (`stops`, `fare_rules`, `stop_times`, `trips`) with two
short-circuit guards. No geometry, date, or graph logic is involved.

## 2. Function Signature

```python
from __future__ import annotations

import polars as pl

from gtfs_validator.context import ValidationContext
from gtfs_validator.notices import Notice, Severity


def validate_stop_zone_id(
    feed: dict[str, pl.DataFrame],
    ctx: ValidationContext,
) -> list[Notice]:
```

The `ctx` argument is unused but required by the standard validator
signature.

## 3. Notice Definitions

### `stop_without_zone_id` (INFO)

Emitted once per stop that:
- has `location_type == 0` (STOP/platform), and
- has `zone_id` null or absent, and
- is served (via `stop_times -> trips -> routes`) by at least one route
  that appears in a `fare_rules` row with a non-null `origin_id`,
  `destination_id`, or `contains_id`.

```python
Notice(
    code="stop_without_zone_id",
    severity=Severity.INFO,
    fields={
        "stop_id": <str>,          # stops.stop_id
        "stop_name": <str | None>, # stops.stop_name (may be null)
        "csv_row_number": <int>,   # stops.csv_row_number (1-based)
    },
)
```

Exactly one notice per violating stop, regardless of how many routes
or fare rules link it to zone-dependent fares.

## 4. Implementation Approach

Use Polars throughout. The core logic is a four-way join chain. Two
early-exit guards avoid any join work when the feed cannot have
violations.

### Pseudocode

```
function validate_stop_zone_id(feed, ctx):

    # Guard 1: fare_rules.txt absent or empty
    if "fare_rules" not in feed or feed["fare_rules"].is_empty():
        return []

    fare_rules = feed["fare_rules"]

    # Identify which zone columns are actually present
    zone_cols = [c for c in ("origin_id", "destination_id", "contains_id")
                 if c in fare_rules.columns]

    if not zone_cols:
        return []

    # Guard 2: hasFareZoneStructure — at least one zone column is non-null somewhere
    has_zone_structure = (
        fare_rules
        .select(pl.any_horizontal([pl.col(c).is_not_null() for c in zone_cols]))
        .to_series()
        .any()
    )
    if not has_zone_structure:
        return []

    # Guard 3: required tables present
    if "stops" not in feed or "stop_times" not in feed or "trips" not in feed:
        return []

    stops     = feed["stops"]
    stop_times = feed["stop_times"]
    trips     = feed["trips"]

    # Build set of route_ids that appear in zone-dependent fare rules.
    # drop_nulls() intentionally excludes global zone rules (route_id null),
    # matching the Java behavior where null route_id never matches a real route.
    zone_fare_routes = (
        fare_rules
        .filter(pl.any_horizontal([pl.col(c).is_not_null() for c in zone_cols]))
        .select("route_id")
        .drop_nulls()
        .unique()
    )

    if zone_fare_routes.is_empty():
        return []

    # Candidate stops: location_type == 0 and zone_id null
    # Retain only columns needed for the join and notice construction.
    required_stop_cols = [c for c in ("stop_id", "stop_name", "csv_row_number")
                          if c in stops.columns]
    candidates = (
        stops
        .filter(
            (pl.col("location_type") == 0) & pl.col("zone_id").is_null()
        )
        .select(required_stop_cols)
    )

    if candidates.is_empty():
        return []

    # Retain only the columns we need from stop_times and trips
    st_cols  = [c for c in ("stop_id", "trip_id") if c in stop_times.columns]
    tr_cols  = [c for c in ("trip_id", "route_id") if c in trips.columns]

    if "stop_id" not in st_cols or "trip_id" not in st_cols:
        return []
    if "trip_id" not in tr_cols or "route_id" not in tr_cols:
        return []

    # Join chain: candidates -> stop_times -> trips -> zone_fare_routes
    violators = (
        candidates
        .join(stop_times.select(st_cols), on="stop_id", how="inner")
        .join(trips.select(tr_cols), on="trip_id", how="inner")
        .join(zone_fare_routes, on="route_id", how="inner")
        .select([c for c in ("stop_id", "stop_name", "csv_row_number")
                 if c in required_stop_cols])
        .unique(subset=["stop_id"])   # one notice per stop (mirrors Java break)
    )

    notices: list[Notice] = []
    for row in violators.iter_rows(named=True):
        notices.append(
            Notice(
                code="stop_without_zone_id",
                severity=Severity.INFO,
                fields={
                    "stop_id": row["stop_id"],
                    "stop_name": row.get("stop_name"),
                    "csv_row_number": row.get("csv_row_number"),
                },
            )
        )
    return notices
```

### Key design decisions

- **Polars join chain, not nested Python loops.** The analysis spec
  confirms this is the idiomatic and efficient translation of the Java
  nested-loop chain (`stop -> stop_times -> trips -> routes ->
  fare_rules match`).

- **`drop_nulls()` on `zone_fare_routes`.** Fare rules with zone
  columns but null `route_id` (global zone rules) are excluded. A null
  `route_id` in `fare_rules` cannot match `trips.route_id` in an inner
  join, so this produces the same result as the Java multimap lookup
  that never finds a match for null keys.

- **`unique(subset=["stop_id"])` for one-notice-per-stop.** Mirrors
  the Java `break` after the first matching route. A stop may appear
  multiple times in the join output if it is served by several routes
  in zone-dependent fare rules; `unique` collapses these to one row.

- **Guard for absent zone columns.** If `fare_rules.txt` exists but
  has none of `origin_id`, `destination_id`, or `contains_id` as
  columns at all (not merely all-null), `zone_cols` is empty and we
  return immediately rather than calling `pl.any_horizontal([])` which
  would raise.

- **`routes.txt` not required.** The Java implementation resolves a
  `GtfsRoute` object, but uses only its `routeId()` string. The Polars
  join goes directly from `trips.route_id` to `fare_rules.route_id`,
  skipping `routes.txt` entirely. The behavior is identical.

- **`csv_row_number` column convention.** Use snake_case
  `csv_row_number` consistent with `stop_name.py`,
  `stop_time_arrival_departure.py`, and other recent validators. Guard
  with `if c in stops.columns` in case the loader omits it.

## 5. Registry Entry

Add to `src/gtfs_validator/validators/__init__.py`:

```python
from gtfs_validator.validators.stop_zone_id import validate_stop_zone_id
```

Add to `VALIDATOR_REGISTRY`:

```python
ValidatorEntry(
    name="stop_zone_id",
    fn=validate_stop_zone_id,
    requires=["stops", "fare_rules", "stop_times", "trips"],
),
```

The `requires` list causes the runner to skip this validator if any of
the four tables is absent from the feed. The internal guards handle
the cases where tables are present but empty or structurally
ineligible (no zone columns used).

`routes` is intentionally omitted from `requires` because the Polars
join does not use `routes.txt`.

## 6. Test Plan

**Test file:** `tests/test_validators/test_stop_zone_id.py`

All tests construct minimal `feed` dicts with Polars DataFrames and
call `validate_stop_zone_id` directly. No file I/O.

### Shared helpers

```python
from datetime import date

import polars as pl
import pytest

from gtfs_validator.context import ValidationContext
from gtfs_validator.notices import Severity
from gtfs_validator.validators.stop_zone_id import validate_stop_zone_id

CTX = ValidationContext(country_code="US", date_for_validation=date(2024, 1, 1))


def make_stops(rows: list[dict]) -> pl.DataFrame:
    schema = {
        "stop_id": pl.Utf8,
        "stop_name": pl.Utf8,
        "location_type": pl.Int64,
        "zone_id": pl.Utf8,
        "csv_row_number": pl.Int64,
    }
    if not rows:
        return pl.DataFrame(schema=schema)
    return pl.DataFrame(rows).cast(schema)


def make_fare_rules(rows: list[dict]) -> pl.DataFrame:
    schema = {
        "route_id": pl.Utf8,
        "origin_id": pl.Utf8,
        "destination_id": pl.Utf8,
        "contains_id": pl.Utf8,
    }
    if not rows:
        return pl.DataFrame(schema=schema)
    return pl.DataFrame(rows).cast(schema)


def make_stop_times(rows: list[dict]) -> pl.DataFrame:
    schema = {"stop_id": pl.Utf8, "trip_id": pl.Utf8}
    if not rows:
        return pl.DataFrame(schema=schema)
    return pl.DataFrame(rows).cast(schema)


def make_trips(rows: list[dict]) -> pl.DataFrame:
    schema = {"trip_id": pl.Utf8, "route_id": pl.Utf8}
    if not rows:
        return pl.DataFrame(schema=schema)
    return pl.DataFrame(rows).cast(schema)


def base_feed(stops=None, fare_rules=None, stop_times=None, trips=None) -> dict:
    """Build a minimal feed with sensible defaults for each table."""
    return {
        "stops": stops if stops is not None else make_stops([]),
        "fare_rules": fare_rules if fare_rules is not None else make_fare_rules([]),
        "stop_times": stop_times if stop_times is not None else make_stop_times([]),
        "trips": trips if trips is not None else make_trips([]),
    }
```
